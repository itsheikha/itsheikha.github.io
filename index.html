// ============================================================ // Cloudflare Worker — مسودتي API Proxy // ============================================================ // Environment Variables (تُضاف في Cloudflare Dashboard): // // NOTION_KEY → Notion Internal Integration Token // DATABASE_ID → Reading Log database ID (سجل القراءات) // THOUGHTS_DB_ID → Thoughts database ID (شذرات) // ADMIN_PASS → كلمة سر بسيطة لحماية النشر // ============================================================ export default { async fetch(request, env) { const url = new URL(request.url); const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json; charset=utf-8", }; if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } const notionHeaders = { Authorization: `Bearer ${env.NOTION_KEY}`, "Notion-Version": "2022-06-28", "Content-Type": "application/json", }; try { // ════════════════════════════════════════════════════════ // GET /api/posts — قائمة المقالات (سجل القراءات) // ════════════════════════════════════════════════════════ if (url.pathname === "/api/posts" && request.method === "GET") { const statusFilter = url.searchParams.get("status"); const body = { sorts: [{ property: "Date", direction: "descending" }], }; if (statusFilter) { body.filter = { property: "Status", status: { equals: statusFilter }, }; } const res = await fetch( `https://api.notion.com/v1/databases/${env.DATABASE_ID}/query`, { method: "POST", headers: notionHeaders, body: JSON.stringify(body) } ); const data = await res.json(); if (!res.ok) return errorResponse(res.status, data, corsHeaders); const posts = (data.results || []).map((p) => simplifyPage(p)); return json({ posts, count: posts.length }, corsHeaders); } // ════════════════════════════════════════════════════════ // GET /api/posts/:id — محتوى مقال واحد // ════════════════════════════════════════════════════════ const postMatch = url.pathname.match( /^\/api\/posts\/([a-f0-9-]{32,36})$/ ); if (postMatch && request.method === "GET") { const pageId = postMatch[1]; const pageRes = await fetch( `https://api.notion.com/v1/pages/${pageId}`, { headers: notionHeaders } ); const pageData = await pageRes.json(); if (!pageRes.ok) return errorResponse(pageRes.status, pageData, corsHeaders); const allBlocks = await fetchAllBlocks(pageId, notionHeaders); for (let i = 0; i < allBlocks.length; i++) { if (allBlocks[i].has_children) { allBlocks[i].children = await fetchAllBlocks(allBlocks[i].id, notionHeaders); } } return json({ meta: simplifyPage(pageData), blocks: allBlocks }, corsHeaders); } // ════════════════════════════════════════════════════════ // GET /api/thoughts — جلب الشذرات // ════════════════════════════════════════════════════════ if (url.pathname === "/api/thoughts" && request.method === "GET") { if (!env.THOUGHTS_DB_ID) { return json({ thoughts: [], count: 0, warning: "THOUGHTS_DB_ID not set" }, corsHeaders); } const res = await fetch( `https://api.notion.com/v1/databases/${env.THOUGHTS_DB_ID}/query`, { method: "POST", headers: notionHeaders, body: JSON.stringify({ sorts: [{ timestamp: "created_time", direction: "descending" }], }), } ); const data = await res.json(); if (!res.ok) return errorResponse(res.status, data, corsHeaders); const thoughts = (data.results || []).map((p) => ({ id: p.id, text: extractTitle(p.properties), date: p.created_time, })); return json({ thoughts, count: thoughts.length }, corsHeaders); } // ════════════════════════════════════════════════════════ // POST /api/thoughts — نشر شذرة جديدة (محمي بكلمة سر) // ════════════════════════════════════════════════════════ if (url.pathname === "/api/thoughts" && request.method === "POST") { if (!env.THOUGHTS_DB_ID) { return json({ error: true, message: "THOUGHTS_DB_ID not configured" }, corsHeaders, 500); } let reqBody; try { reqBody = await request.json(); } catch { return json({ error: true, message: "Invalid JSON body" }, corsHeaders, 400); } // ── التحقق من كلمة السر ────────────────────────────── if (!reqBody.password || reqBody.password !== env.ADMIN_PASS) { return json({ error: true, message: "كلمة السر غير صحيحة" }, corsHeaders, 401); } const text = (reqBody.text || "").trim(); if (!text) { return json({ error: true, message: "النص مطلوب" }, corsHeaders, 400); } // ── إنشاء صفحة جديدة في Notion ────────────────────── const createRes = await fetch("https://api.notion.com/v1/pages", { method: "POST", headers: notionHeaders, body: JSON.stringify({ parent: { database_id: env.THOUGHTS_DB_ID }, properties: { title: { title: [{ text: { content: text } }], }, }, }), }); const createData = await createRes.json(); if (!createRes.ok) return errorResponse(createRes.status, createData, corsHeaders); return json({ success: true, thought: { id: createData.id, text: text, date: createData.created_time, }, }, corsHeaders); } // ════════════════════════════════════════════════════════ // GET /api/schema — هيكل قاعدة البيانات (للتشخيص) // ════════════════════════════════════════════════════════ if (url.pathname === "/api/schema") { const dbId = url.searchParams.get("db") === "thoughts" ? env.THOUGHTS_DB_ID : env.DATABASE_ID; const res = await fetch( `https://api.notion.com/v1/databases/${dbId}`, { headers: notionHeaders } ); const data = await res.json(); if (!res.ok) return errorResponse(res.status, data, corsHeaders); const schema = {}; for (const [key, val] of Object.entries(data.properties || {})) { const entry = { type: val.type }; if (val.type === "status") entry.options = (val.status?.options || []).map((o) => o.name); if (val.type === "select") entry.options = (val.select?.options || []).map((o) => o.name); schema[key] = entry; } return json({ title: data.title?.[0]?.plain_text || "", schema }, corsHeaders); } // ════════════════════════════════════════════════════════ // GET / — الصفحة الرئيسية // ════════════════════════════════════════════════════════ if (url.pathname === "/") { return json({ name: "مسودتي API", endpoints: [ "GET /api/posts — reading log", "GET /api/posts?status=Done — filtered", "GET /api/posts/:id — post + blocks", "GET /api/thoughts — all thoughts", "POST /api/thoughts — create thought", "GET /api/schema — db structure", "GET /api/schema?db=thoughts — thoughts db structure", ], }, corsHeaders); } return json({ error: true, message: "Not found" }, corsHeaders, 404); } catch (err) { return json({ error: true, message: err.message || "Internal error" }, corsHeaders, 500); } }, }; // ══════════════════════════════════════════════════════════════ // Helpers // ══════════════════════════════════════════════════════════════ function json(data, headers, status = 200) { return new Response(JSON.stringify(data), { status, headers }); } function errorResponse(status, data, headers) { return new Response( JSON.stringify({ error: true, message: data.message || "Notion error", code: data.code }), { status, headers } ); } async function fetchAllBlocks(blockId, headers) { let all = [], cursor, hasMore = true; while (hasMore) { const u = new URL(`https://api.notion.com/v1/blocks/${blockId}/children`); u.searchParams.set("page_size", "100"); if (cursor) u.searchParams.set("start_cursor", cursor); const res = await fetch(u.toString(), { headers }); const data = await res.json(); all = all.concat(data.results || []); hasMore = data.has_more || false; cursor = data.next_cursor; } return all; } function extractTitle(props) { for (const [, val] of Object.entries(props)) { if (val.type === "title") return val.title?.[0]?.plain_text || ""; } return ""; } function extractDate(props) { if (props.Date?.type === "date") return props.Date.date?.start || ""; for (const [, val] of Object.entries(props)) { if (val.type === "date") return val.date?.start || ""; } return ""; } function extractURL(props) { if (props.URL?.type === "url") return props.URL.url || ""; for (const [, val] of Object.entries(props)) { if (val.type === "url") return val.url || ""; } return ""; } function extractStatus(props) { if (props.Status?.type === "status") return props.Status.status?.name || ""; if (props.Status?.type === "select") return props.Status.select?.name || ""; return ""; } function simplifyPage(page) { const props = page.properties || {}; return { id: page.id, title: extractTitle(props), date: extractDate(props), url: extractURL(props), status: extractStatus(props), quote: props.Quote?.rich_text?.[0]?.plain_text || "", created: page.created_time, edited: page.last_edited_time, }; }
